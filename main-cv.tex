\section*{Research Experience (continued)}

\begin{cvEntries}

    % ---------- PRD Research ----------

    \researchEntry
    {Residual Galactic Binary Foregrounds
    in LISA SGWB Inference}
    {National Gravitation Laboratory and
    School of Physics, HUST}
    {2026}

    \begin{cvBullets}

        \item
        Estimated the residual Galactic-binary
        power spectrum using injection--recovery
        catalogs.

        \item
        Quantified non-Gaussianity from source
        discreteness using excess kurtosis.
        Instrumental noise and SGWB power
        largely suppress this signature.

        \item
        Assessed spectral degeneracy with an
        isotropic SGWB using Fisher analysis.
        Marginalizing over the residual-power
        normalization increased the SGWB-amplitude
        uncertainty by 13.6\% in the fiducial
        Gaussian-covariance model.

    \end{cvBullets}

    % ---------- JHEAp Research ----------

    \researchEntry
    {Time-Series and Correlation Analysis
    of Gamma-Ray Burst Light Curves}
    {Department of Astronomy, HUST}
    {Oct.\ 2024 -- Jan.\ 2026}

    \begin{cvBullets}

        \item
        Measured Hurst indices for 163
        long-duration BATSE gamma-ray bursts
        using detrended fluctuation analysis.

        \item
        Found anti-correlations with burst
        durations and moderate positive correlations
        with peak photon flux. Standard spectral
        parameters showed no significant
        linear correlations.

    \end{cvBullets}

    % ---------- M.Sc. Research ----------

    \researchEntry
    {Identifying Sources in Cygnus OB2
    Using Multiwavelength Observations}
    {Department of Physics, HKU}
    {Sep.\ 2023 -- Jun.\ 2024}

    \begin{cvBullets}

        \item
        Analyzed Chandra X-ray observations
        of Cygnus OB2 and cross-matched
        the detected sources with Gaia.

        \item
        Identified optical counterparts for
        106 of 148 X-ray sources.
        Hardness ratios suggested that
        12 unmatched sources might be
        background galaxies.

    \end{cvBullets}

    % ---------- A&A Research ----------

    \researchEntry
    {Gamma-Ray Burst Prompt Emission
    and Polarization}
    {Center for Theoretical Physics and
    College of Physics, Jilin University}
    {2022}

    \begin{cvBullets}

        \item
        Predicted upper limits on time-integrated
        polarization for 37 GRBs using synchrotron
        emission in ordered magnetic fields.

        \item
        Most polarization observations were
        consistent with the model, while two
        bursts exceeded the predicted limits.

        \item
        Predicted energy-dependent polarization
        signatures for POLAR-2 to test competing
        prompt-emission models.

    \end{cvBullets}

\end{cvEntries}
